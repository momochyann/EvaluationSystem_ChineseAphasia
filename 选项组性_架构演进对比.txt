        if (!所有选项组.Contains(this))
        {
            所有选项组.Add(this);
        }

v3.0 (System)：

    {
        this.GetSystem<选项组高亮管理System>().注册选项组(this);
    }

     ✅ System 内部处理重复检查


场景 2: 通知其他组取消高亮
────────────────────────────────────────────────────────────────

v2.0 (静态列表)：
    private void 通知其他选项组取消高亮()
    {
        foreach (var 其他选项组 in 所有选项组)
        {
            if (其他选项组 == this) continue;
            if (其他选项组.IsAnimationOnSelect)
            {
                其他选项组.取消高亮();
            }
        }
    }

v3.0 (System)：
    // 只需一行！
    this.GetSystem<选项组高亮管理System>().设置高亮组(this);

优势：✅ 代码更简洁
     ✅ 逻辑集中在 System
     ✅ 更容易理解和维护

     ✅ System 维护状态

    ❌ 需要在每个选项组中添加历史记录代码

v3.0 实现：

        private Stack<选项组性_茉> 历史栈 = new Stack<选项组性_茉>();

        {
            if (历史栈.Count > 0)
            {
                var 上一个组 = 历史栈.Pop();
                设置高亮组(上一个组);
            }
        }
    
    ✅ 选项组代码无需修改！
    ✅ 一处修改，全局生效

v3.0 实现：

        private Dictionary<string, int> 使用统计 = new Dictionary<string, int>();
        
        public void 设置高亮组(选项组性_茉 目标选项组)
        {
            // ... 原有逻辑 ...
            
            // 统计使用次数

                使用统计[目标选项组.选项组名称] = 0;
            使用统计[目标选项组.选项组名称]++;
        }
    
    ✅ 统计逻辑集中管理
